⚡ COMMANDES QUOTIDIENNES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Gestion des services :
   
      make start          # Démarrer
      make stop           # Arrêter
      make restart        # Redémarrer
      make status         # Statut
      make logs           # Voir logs
   
   ─────────────────────────────────────────────────────────────────────
   
   Base de données :
   
      make migrate        # Migrer
      make seed           # Données test
      make fresh          # Reset + migrate + seed
      make db-backup      # Backup
      make db-cli         # MySQL CLI
   
   ─────────────────────────────────────────────────────────────────────
   
   Créer un client (tenant) :
   
      make tenant-create name="nom-client"
      make tenant-list
   
   ─────────────────────────────────────────────────────────────────────
   
   Développement :
   
      make bash-backend       # Terminal backend
      make bash-frontend      # Terminal frontend
      make artisan cmd="..."  # Commande artisan
      make npm cmd="..."      # Commande npm
   
   ─────────────────────────────────────────────────────────────────────
   
   Aide :
   
      make help           # Toutes les commandes
      make urls           # Toutes les URLs

🚨 DÉPANNAGE RAPIDE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   Les services ne démarrent pas :
      make clean && make install && make start
   
   Erreur de permissions :
      make fix-permissions
   
   Base de données corrompue :
      make db-reset
   
   Problème de cache :
      make clear-cache
   
   Tout réinitialiser :
      make reset
